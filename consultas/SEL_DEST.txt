SELECT 
    F.CNPJ     AS CNPJ,
    F.RAZAOSOC AS RAZAO_SOCIAL,
    F.INSCEST  AS INSCRICAO_ESTADUAL,
    F.EMAIL    AS EMAIL_CONTATO,
    F.ENDERECO AS LOGRADOURO,
    F.NUMEEND  AS NUMERO,
    F.COMPEND  AS COMPLEMENTO,
    F.BAIRRO   AS BAIRRO,
    F.CIDADE   AS COD_MUNICIPIO,
    C.NOME     AS MUNICIPIO,
    C.UF       AS UF,
    F.CEP      AS CEP,
    F.TELEFONE AS TELEFONE
  FROM FILIAL      F
 INNER JOIN CIDADE C ON C.CIDADE = F.CIDADE
 WHERE ESTAB = :ESTAB
